#Estimacion talla media

talla_media_mes = function(arreglo, mes, sexo, puerto, arte){
  
  
  tabla = table(arreglo$ANCHO_MM[arreglo$SEXO == sexo & arreglo$MES == mes &
                                   arreglo$PUERTO_RECALADA == puerto &
                                   arreglo$ARTE == arte])
  prop = numeric(length(tabla))
  mult = numeric(length(tabla))
  talla <- as.numeric(row.names(tabla))
  
  tabla = as.matrix(tabla)
  suma = sum(tabla)
  
  for (i in 1:length(tabla)) {
    
    prop[i] = tabla[i]/suma
    
  }
  
  
  for (i in 1:length(tabla)) {
    
    mult[i] = talla[i]*prop[i]
    
  }
  
  sum(mult)
}

BootStrap_tallamedia = function(arreglo, sexo, puerto, arte, n){
  
  
  tabla = arreglo[arreglo$SEXO == sexo & arreglo$PUERTO_RECALADA == puerto & arreglo$ARTE == arte,]
  
  tab_var = numeric(0)
  
  for (i in 1:12 ) {
    
    tab_boot = numeric(0)
    tab_par = numeric(0)
    
    tab_aux = tabla[tabla$MES == i,]
    
    for(j in 1:n){
      
      #tab_boot = tab_aux[sample(1:nrow(tab_aux), size = nrow(tab_aux), replace = TRUE), ]
      tab_boot = tab_aux %>%
        group_by(VIAJE) %>%
        group_modify(~ {
          df <- .x
          df[sample.int(nrow(df), nrow(df), replace = TRUE), ]
        }) %>%
        ungroup()
      tab_par[j] = talla_media_mes(tab_boot, i, sexo, puerto, arte)
      
      
    }
    tab_var[i] = var(tab_par)
    
  }
  
  return(tab_var)
  
}

talla_media_mes = function(arreglo, mes, sexo, puerto, arte){
  
  
  tabla = table(arreglo$ANCHO_MM[arreglo$SEXO == sexo & arreglo$MES == mes &
                                   arreglo$PUERTO_RECALADA == puerto &
                                   arreglo$ARTE == arte])
  prop = numeric(length(tabla))
  mult = numeric(length(tabla))
  talla <- as.numeric(row.names(tabla))
  
  tabla = as.matrix(tabla)
  suma = sum(tabla)
  
  for (i in 1:length(tabla)) {
    
    prop[i] = tabla[i]/suma
    
  }
  
  
  for (i in 1:length(tabla)) {
    
    mult[i] = talla[i]*prop[i]
    
  }
  
  sum(mult)
}

Var_est_talla_media = function(arreglo, sexo, puerto, arte, mes){
  
  tabla = arreglo[arreglo$SEXO == sexo & arreglo$PUERTO_RECALADA == puerto &
                    arreglo$ARTE == arte & arreglo$MES == mes,]
  
  tabla = tabla %>%
    filter(!is.na(COD_BARCO)) %>%
    group_by(VIAJE) %>%
    mutate(ejemplar = row_number()) %>%
    ungroup() 
  
  
  if (length(unique(tabla$VIAJE)) == 1 || arte == 1) {
    diseno <- svydesign(
      id = ~ejemplar,
      data = tabla,
    )
    
  }else{
    diseno <- svydesign(
      id = ~VIAJE + ejemplar,
      data = tabla,
    )
  }
  SE(svymean(~ANCHO_MM, diseno))[1]^2
}

talla_media_mes = function(arreglo, mes, sexo, puerto, arte){
  
  
  tabla = table(arreglo$ANCHO_MM[arreglo$SEXO == sexo & arreglo$MES == mes &
                                   arreglo$PUERTO_RECALADA == puerto &
                                   arreglo$ARTE == arte])
  prop = numeric(length(tabla))
  mult = numeric(length(tabla))
  talla <- as.numeric(row.names(tabla))
  
  tabla = as.matrix(tabla)
  suma = sum(tabla)
  
  for (i in 1:length(tabla)) {
    
    prop[i] = tabla[i]/suma
    
  }
  
  
  for (i in 1:length(tabla)) {
    
    mult[i] = talla[i]*prop[i]
    
  }
  
  sum(mult)
}

    tab_aux = tabla[tabla$MES == i,]

  }
  
  return(tab_var)

RenBuc <- function(DATA){
  n <- nrow(DATA)
  d <- DATA$DESEMBARQUE
  e <- DATA$HORAS
  
  U <- sum(d)/sum(e)
  V <- (1/(n*(mean(e)^{2})))*(sum((d-U*e)^{2}))/(n-1)
  
  cat("El rendimiento estimado es:", U," kg jaiba/hora buceo\n")
  cat("La variabilidad estimada del rendimiento estimado es:", V)
}

RenTram <- function(DATA){
  n <- nrow(DATA)
  d <- DATA$DESEMBARQUE
  e <- DATA$N
  
  U <- sum(d)/sum(e)
  V <- (1/(n*(mean(e)^{2})))*(sum((d-U*e)^{2}))/(n-1)
  
  cat("El rendimiento estimado es:", U," kg jaiba/trampa\n")
  cat("La variabilidad estimada del rendimiento estimado es:", V)
}

BootStrap <- function(d,e,B){
  n <- length(d)
  resultados <- numeric(B)
  for (b in 1:B) {
    i <- sample(1:n,n,replace = T)
    d_boot <- d[i]
    e_boot <- e[i]
    resultados[b] <- sum(d_boot)/sum(e_boot)
  }
  return(resultados)
}

RenReg <- function(data) {
  # Asegurar que las variables existan
  if(!all(c("DESEMBARQUE", "N") %in% names(data))) {
    stop("El data frame debe contener las columnas 'DESEMBARQUE' y 'N'")
  }
  
  # Ajuste del modelo lineal
  m1 <- lm(DESEMBARQUE ~ N, data = data)
  
  # Extraer parámetros
  b0 <- coef(m1)[1]
  b1 <- coef(m1)[2]
  
  # Número de observaciones
  n <- nrow(data)
  
  # Estimación de sigma^2
  sigma2est <- sum(resid(m1)^2) / (n - 2)
  
  # Estadísticos de la variable N
  xmedia <- mean(data$N)
  sxx <- sum((data$N - xmedia)^2)
  z <- n / sum(data$N)
  
  # Varianzas y covarianza de los estimadores
  varb0 <- sigma2est * (1/n + (xmedia^2 / sxx))
  varb1 <- sigma2est / sxx
  covb0b1 <- -sigma2est * xmedia / sxx
  
  # Estimador de rendimiento U y su varianza
  U <- b1 + z * b0
  varU <- sigma2est * ((1 - z * xmedia)^2 / sxx + (z^2) / n)
  
  # Salida
  cat("========================================\n")
  cat("   ESTIMADOR DE REGRESIÓN (Cochran)\n")
  cat("========================================\n\n")
  cat("Coeficientes estimados:\n")
  cat(sprintf("  b0 (Intercepto): %.6f\n", b0))
  cat(sprintf("  b1 (Pendiente):  %.6f\n\n", b1))
  
  cat("Varianzas y covarianza:\n")
  cat(sprintf("  Var(b0): %.6f\n", varb0))
  cat(sprintf("  Var(b1): %.6f\n", varb1))
  cat(sprintf("  Cov(b0, b1): %.6f\n\n", covb0b1))
  
  cat("Estimación del rendimiento:\n")
  cat(sprintf("  U estimado (kg jaiba/trampa): %.6f\n", U))
  cat(sprintf("  Var(U estimado): %.6f\n\n", varU))
  
  cat("========================================\n")
  
  # Retornar invisiblemente una lista con todos los resultados
  invisible(list(
    modelo = m1,
    b0 = b0,
    b1 = b1,
    varb0 = varb0,
    varb1 = varb1,
    covb0b1 = covb0b1,
    U = U,
    varU = varU,
    sigma2 = sigma2est,
    z = z
  ))
}

RenSchaefer <- function(data, N_var, Y_var, alpha = 0.05, plot = TRUE) {
  # data  : data.frame con los datos
  # N_var : nombre (string) de la variable esfuerzo (ej. "N")
  # Y_var : nombre (string) de la variable desembarque (ej. "DESEMBARQUE")
  # alpha : nivel para IC (por defecto 0.05 => IC 95%)
  # plot  : si TRUE dibuja la nube y la curva ajustada
  
  # Extraer
  E <- data[[N_var]]
  Y <- data[[Y_var]]
  
  if (any(E == 0)) stop("Hay valores de esfuerzo (N) iguales a 0. El modelo no puede ajustarse.")
  n <- length(E)
  
  # Starts sencillos: Uf = Y/E = a - b E
  Uf <- Y / E
  lm_R <- lm(Uf ~ E)
  a_start <- coef(lm_R)[1]
  b_start <- -coef(lm_R)[2]
  
  # Ajuste nls con control
  # uso try() por si falla la convergencia, devolver error claro
  fit_try <- try(nls(Y ~ a * E - b * E^2,
                     start = list(a = a_start, b = b_start),
                     control = nls.control(maxiter = 1000, warnOnly = TRUE)),
                 silent = TRUE)
  if (inherits(fit_try, "try-error")) stop("El ajuste nls falló. Intenta otros valores iniciales o nlsLM.")
  modelo <- fit_try
  
  # Coeficientes
  coefs <- coef(modelo)
  a <- as.numeric(coefs["a"])
  b <- as.numeric(coefs["b"])
  
  # Rendimiento estimado (total tipo RenTram)
  Y_est <- a * E - b * E^2
  R_total <- sum(Y_est) / sum(E)
  
  # Varianza del estimador por delta method usando vcov()
  V <- vcov(modelo)       # 2x2
  # derivadas
  dR_da <- 1
  dR_db <- - sum(E^2) / sum(E)
  var_R <- as.numeric(dR_da^2 * V["a","a"] +
                        dR_db^2 * V["b","b"] +
                        2 * dR_da * dR_db * V["a","b"])
  sd_R <- sqrt(var_R)
  
  # IC analítico (t con df = n - p)
  p <- length(coefs)      # 2
  df <- n - p
  tcrit <- qt(1 - alpha/2, df = df)
  CI_analitico <- R_total + c(-1,1) * tcrit * sd_R
  
  # Plot opcional
  if (plot) {
    plot(E, Y,
         xlab = "Esfuerzo (N trampas)",
         ylab = "Desembarque (kg)",
         main = "Curva de rendimiento tipo Schaefer")
    E_seq <- seq(min(E), max(E), length.out = 200)
    Y_pred <- a * E_seq - b * E_seq^2
    lines(E_seq, Y_pred, col = "blue", lwd = 2)
  }
  
  # Resumen por consola
  cat("\n--- RESUMEN DEL MODELO DE SCHAEFER ---\n")
  cat(sprintf("Parámetro a (tasa inicial): %.6f\n", a))
  cat(sprintf("Parámetro b (curvatura):   %.6f\n", b))
  cat(sprintf("Rendimiento estimado R_hat: %.6f (kg por unidad de esfuerzo)\n", R_total))
  cat(sprintf("Varianza estimada de R_hat: %.8f\n", var_R))
  cat(sprintf("Desviación estándar (sd):   %.6f\n", sd_R))
  cat(sprintf("IC analítico %.1f%% = [%.6f, %.6f] (t, df=%d)\n",
              100*(1-alpha), CI_analitico[1], CI_analitico[2], df))
  cat("----------------------------------------\n")
  cat("Interpretación breve:\n")
  cat(" - 'a' es el rendimiento por unidad de esfuerzo cuando E es muy pequeño.\n")
  cat(" - 'b' mide la caída del rendimiento al aumentar el esfuerzo (mayor b -> más saturación).\n")
  cat(" - La varianza indicada es la estimación (delta method) de la varianza de R_hat.\n\n")
  
  # Devolver lista con todo
  invisible(list(
    model_summary = summary(modelo),
    coeficientes = coefs,
    R_hat = R_total,
    var_R = var_R,
    sd_R = sd_R,
    CI = CI_analitico,
    df = df,
    vcov = V
  ))
}

RenRegMulti <- function(data, xname = "N", covariates = NULL,
                        alpha = 0.05, boot = FALSE, B = 1000, seed = 123) {
  # data: data.frame con columnas DESEMBARQUE y xname y covariates si los hay
  # xname: nombre (string) de la variable esfuerzo, p.ej. "N"
  # covariates: vector de nombres (strings) de variables auxiliares, p.ej. c("MES")
  #   las variables en 'covariates' se incluirán como factores automáticamente
  # boot: si TRUE, calcula IC bootstrap (percentil) con B réplicas
  # devuelve lista invisible con resultados y muestra un resumen tipo RenTram
  
  # chequeos
  if (!("DESEMBARQUE" %in% names(data))) stop("data debe tener columna 'DESEMBARQUE'")
  if (!(xname %in% names(data))) stop(paste0("data debe tener columna '", xname, "'"))
  n <- nrow(data)
  
  # construir fórmula
  formula_parts <- xname
  if (!is.null(covariates)) {
    # convertir covariates a factor en la fórmula
    cov_terms <- sapply(covariates, function(v) paste0("factor(", v, ")"))
    formula_parts <- c(formula_parts, cov_terms)
  }
  fmla <- as.formula(paste("DESEMBARQUE ~", paste(formula_parts, collapse = " + ")))
  
  # ajustar modelo
  m <- lm(fmla, data = data)
  coef_hat <- coef(m)
  p <- length(coef_hat)
  
  # matriz de diseño y sumas de columnas (coincide con coef names)
  MM <- model.matrix(m)       # n x p
  sums_cols <- colSums(MM)    # vector longitud p, nombre coincide con coef names
  sum_x <- sum(data[[xname]])
  
  # vector 'a' para delta method: a_j = sum(column_j) / sum_x
  a_vec <- as.numeric(sums_cols / sum_x)
  names(a_vec) <- names(coef_hat)
  
  # cov matrix of coef
  Vcoef <- vcov(m)            # p x p
  # estimated U and var(U)
  U_hat <- sum(sums_cols * coef_hat) / sum_x     # = t(a_vec) %*% coef_hat
  varU_hat <- as.numeric(t(a_vec) %*% Vcoef %*% a_vec)
  seU_hat <- sqrt(varU_hat)
  
  # degrees freedom and t critical
  df <- n - p
  tcrit <- qt(1 - alpha/2, df = df)
  CI_analitico <- U_hat + c(-1,1) * tcrit * seU_hat
  
  # bootstrap opcional: remuestrear filas, recalcular (coef, sums, U_boot)
  CI_boot <- NULL
  meanUboot <- sdUboot <- NA
  if (boot) {
    set.seed(seed)
    Uboot <- numeric(B)
    for (b in seq_len(B)) {
      i <- sample.int(n, size = n, replace = TRUE)
      dboot <- data[i, , drop = FALSE]
      mboot <- lm(fmla, data = dboot)
      # recomputar sums_cols_boot (usar model.matrix de la muestra bootstrap)
      MMb <- model.matrix(mboot)
      # Es importante alinear nombres: si alguna categoría no aparece en bootstrap,
      # las columnas pueden cambiar. Para robustez, usaremos coefs y sums según mm colnames.
      coefs_b <- coef(mboot)
      sums_b <- colSums(MMb)
      sum_x_b <- sum(dboot[[xname]])
      Uboot[b] <- sum(sums_b * coefs_b) / sum_x_b
    }
    CI_boot <- quantile(Uboot, probs = c(alpha/2, 1 - alpha/2), na.rm = TRUE)
    meanUboot <- mean(Uboot, na.rm = TRUE)
    sdUboot <- sd(Uboot, na.rm = TRUE)
  }
  
  # impresión tipo RenTram
  cat("========================================\n")
  cat("  ESTIMADOR DE REGRESIÓN MULTIPLE (Cochran extendido)\n")
  cat("========================================\n\n")
  cat("Fórmula ajustada:\n")
  cat(deparse(fmla), "\n\n")
  
  cat("Coeficientes estimados (coef_hat):\n")
  print(round(coef_hat, 6))
  cat("\nMatriz de var-cov de coeficientes (vcov):\n")
  print(round(Vcoef, 4))
  cat("\nSumas de columnas de la matriz de diseño (s_j):\n")
  print(round(sums_cols, 4))
  cat("\nVector a (s_j / sum(x)):\n")
  print(round(a_vec, 6))
  cat("\nEstimación del rendimiento:\n")
  cat(sprintf("  U estimado (kg jaiba/%s): %.6f\n", xname, U_hat))
  cat(sprintf("  Var(U estimado): %.6f\n", varU_hat))
  cat(sprintf("  se(U): %.6f\n\n", seU_hat))
  cat(sprintf("  IC analítico %.1f%% = [%.6f, %.6f] (t, df=%d)\n",
              100*(1-alpha), CI_analitico[1], CI_analitico[2], df))
  if (boot) {
    cat(sprintf("  IC bootstrap (percentil %.1f%%) = [%.6f, %.6f]\n",
                100*(1-alpha), CI_boot[1], CI_boot[2]))
    cat(sprintf("  mean(Uboot)=%.6f  sd(Uboot)=%.6f (B=%d)\n", meanUboot, sdUboot, B))
  }
  cat("\n========================================\n")
  
  # devolver resultados
  invisible(list(
    model = m,
    coef = coef_hat,
    vcov = Vcoef,
    sums_cols = sums_cols,
    a = a_vec,
    U = U_hat,
    varU = varU_hat,
    seU = seU_hat,
    CI_analitico = CI_analitico,
    CI_boot = CI_boot,
    Uboot_mean = meanUboot,
    Uboot_sd = sdUboot,
    df = df
  ))
}

b_boot <- numeric(B)

for (i in 1:B) {
  idx <- sample(seq_along(E), replace = TRUE)
  E_b <- E[idx]; Y_b <- Y[idx]
  
  try({
    fit_b <- nls(Y_b ~ a * E_b - b * E_b^2, start = list(a = a, b = b))
    coef_b <- coef(fit_b)
    a_boot[i] <- coef_b["a"]
    b_boot[i] <- coef_b["b"]
    
    Y_est_b <- coef_b["a"] * E_b - coef_b["b"] * E_b^2
    R_total_b <- sum(Y_est_b) / sum(E_b)
    R_boot[i] <- R_total_b
  }, silent = TRUE)
}

plot_desembarque <- function(data, xvar = "N", yvar = "DESEMBARQUE") {
  # Calcular correlación
  r <- cor(data[[xvar]], data[[yvar]], use = "complete.obs", method = "pearson")
  
  # Crear texto de leyenda
  corr_label <- paste0("r = ", round(r, 2))
  
  # Generar gráfico
  ggplot(data, aes_string(x = xvar, y = yvar)) +
    geom_point(color = "darkblue", size = 2) +
    labs(x = "Esfuerzo (horas)",
         y = "Desembarque (kg jaiba)") +
    annotate("text",
             x = Inf, y = Inf,
             label = corr_label,
             hjust = 1.1, vjust = 2,
             color = "black", size = 7) +
    theme_bw() +
    theme(
      legend.position = "none",
      plot.title = element_blank(),
      plot.subtitle = element_blank()
    )
}

RenBuc <- function(DATA){
  n <- nrow(DATA)
  d <- DATA$DESEMBARQUE
  e <- DATA$HORAS
  
  U <- sum(d)/sum(e)
  V <- (1/(n*(mean(e)^{2})))*(sum((d-U*e)^{2}))/(n-1)
  
  cat("El rendimiento estimado es:", U," kg jaiba/hora buceo\n")
  cat("La variabilidad estimada del rendimiento estimado es:", V)
}

RenTram <- function(DATA){
  n <- nrow(DATA)
  d <- DATA$DESEMBARQUE
  e <- DATA$N
  
  U <- sum(d)/sum(e)
  V <- (1/(n*(mean(e)^{2})))*(sum((d-U*e)^{2}))/(n-1)
  
  cat("El rendimiento estimado es:", U," kg jaiba/trampa\n")
  cat("La variabilidad estimada del rendimiento estimado es:", V)
}

BootStrap <- function(d,e,B){
  n <- length(d)
  resultados <- numeric(B)
  for (b in 1:B) {
    i <- sample(1:n,n,replace = T)
    d_boot <- d[i]
    e_boot <- e[i]
    resultados[b] <- sum(d_boot)/sum(e_boot)
  }
  return(resultados)
}

RenReg <- function(data) {
  # Asegurar que las variables existan
  if(!all(c("DESEMBARQUE", "N") %in% names(data))) {
    stop("El data frame debe contener las columnas 'DESEMBARQUE' y 'N'")
  }
  
  # Ajuste del modelo lineal
  m1 <- lm(DESEMBARQUE ~ N, data = data)
  
  # Extraer parámetros
  b0 <- coef(m1)[1]
  b1 <- coef(m1)[2]
  
  # Número de observaciones
  n <- nrow(data)
  
  # Estimación de sigma^2
  sigma2est <- sum(resid(m1)^2) / (n - 2)
  
  # Estadísticos de la variable N
  xmedia <- mean(data$N)
  sxx <- sum((data$N - xmedia)^2)
  z <- n / sum(data$N)
  
  # Varianzas y covarianza de los estimadores
  varb0 <- sigma2est * (1/n + (xmedia^2 / sxx))
  varb1 <- sigma2est / sxx
  covb0b1 <- -sigma2est * xmedia / sxx
  
  # Estimador de rendimiento U y su varianza
  U <- b1 + z * b0
  varU <- sigma2est * ((1 - z * xmedia)^2 / sxx + (z^2) / n)
  
  # Salida
  cat("========================================\n")
  cat("   ESTIMADOR DE REGRESIÓN (Cochran)\n")
  cat("========================================\n\n")
  cat("Coeficientes estimados:\n")
  cat(sprintf("  b0 (Intercepto): %.6f\n", b0))
  cat(sprintf("  b1 (Pendiente):  %.6f\n\n", b1))
  
  cat("Varianzas y covarianza:\n")
  cat(sprintf("  Var(b0): %.6f\n", varb0))
  cat(sprintf("  Var(b1): %.6f\n", varb1))
  cat(sprintf("  Cov(b0, b1): %.6f\n\n", covb0b1))
  
  cat("Estimación del rendimiento:\n")
  cat(sprintf("  U estimado (kg jaiba/trampa): %.6f\n", U))
  cat(sprintf("  Var(U estimado): %.6f\n\n", varU))
  
  cat("========================================\n")
  
  # Retornar invisiblemente una lista con todos los resultados
  invisible(list(
    modelo = m1,
    b0 = b0,
    b1 = b1,
    varb0 = varb0,
    varb1 = varb1,
    covb0b1 = covb0b1,
    U = U,
    varU = varU,
    sigma2 = sigma2est,
    z = z
  ))
}

RenSchaefer <- function(data, N_var, Y_var, alpha = 0.05, plot = TRUE) {
  # data  : data.frame con los datos
  # N_var : nombre (string) de la variable esfuerzo (ej. "N")
  # Y_var : nombre (string) de la variable desembarque (ej. "DESEMBARQUE")
  # alpha : nivel para IC (por defecto 0.05 => IC 95%)
  # plot  : si TRUE dibuja la nube y la curva ajustada
  
  # Extraer
  E <- data[[N_var]]
  Y <- data[[Y_var]]
  
  if (any(E == 0)) stop("Hay valores de esfuerzo (N) iguales a 0. El modelo no puede ajustarse.")
  n <- length(E)
  
  # Starts sencillos: Uf = Y/E = a - b E
  Uf <- Y / E
  lm_R <- lm(Uf ~ E)
  a_start <- coef(lm_R)[1]
  b_start <- -coef(lm_R)[2]
  
  # Ajuste nls con control
  # uso try() por si falla la convergencia, devolver error claro
  fit_try <- try(nls(Y ~ a * E - b * E^2,
                     start = list(a = a_start, b = b_start),
                     control = nls.control(maxiter = 1000, warnOnly = TRUE)),
                 silent = TRUE)
  if (inherits(fit_try, "try-error")) stop("El ajuste nls falló. Intenta otros valores iniciales o nlsLM.")
  modelo <- fit_try
  
  # Coeficientes
  coefs <- coef(modelo)
  a <- as.numeric(coefs["a"])
  b <- as.numeric(coefs["b"])
  
  # Rendimiento estimado (total tipo RenTram)
  Y_est <- a * E - b * E^2
  R_total <- sum(Y_est) / sum(E)
  
  # Varianza del estimador por delta method usando vcov()
  V <- vcov(modelo)       # 2x2
  # derivadas
  dR_da <- 1
  dR_db <- - sum(E^2) / sum(E)
  var_R <- as.numeric(dR_da^2 * V["a","a"] +
                        dR_db^2 * V["b","b"] +
                        2 * dR_da * dR_db * V["a","b"])
  sd_R <- sqrt(var_R)
  
  # IC analítico (t con df = n - p)
  p <- length(coefs)      # 2
  df <- n - p
  tcrit <- qt(1 - alpha/2, df = df)
  CI_analitico <- R_total + c(-1,1) * tcrit * sd_R
  
  # Plot opcional
  if (plot) {
    plot(E, Y,
         xlab = "Esfuerzo (N trampas)",
         ylab = "Desembarque (kg)",
         main = "Curva de rendimiento tipo Schaefer")
    E_seq <- seq(min(E), max(E), length.out = 200)
    Y_pred <- a * E_seq - b * E_seq^2
    lines(E_seq, Y_pred, col = "blue", lwd = 2)
  }
  
  # Resumen por consola
  cat("\n--- RESUMEN DEL MODELO DE SCHAEFER ---\n")
  cat(sprintf("Parámetro a (tasa inicial): %.6f\n", a))
  cat(sprintf("Parámetro b (curvatura):   %.6f\n", b))
  cat(sprintf("Rendimiento estimado R_hat: %.6f (kg por unidad de esfuerzo)\n", R_total))
  cat(sprintf("Varianza estimada de R_hat: %.8f\n", var_R))
  cat(sprintf("Desviación estándar (sd):   %.6f\n", sd_R))
  cat(sprintf("IC analítico %.1f%% = [%.6f, %.6f] (t, df=%d)\n",
              100*(1-alpha), CI_analitico[1], CI_analitico[2], df))
  cat("----------------------------------------\n")
  cat("Interpretación breve:\n")
  cat(" - 'a' es el rendimiento por unidad de esfuerzo cuando E es muy pequeño.\n")
  cat(" - 'b' mide la caída del rendimiento al aumentar el esfuerzo (mayor b -> más saturación).\n")
  cat(" - La varianza indicada es la estimación (delta method) de la varianza de R_hat.\n\n")
  
  # Devolver lista con todo
  invisible(list(
    model_summary = summary(modelo),
    coeficientes = coefs,
    R_hat = R_total,
    var_R = var_R,
    sd_R = sd_R,
    CI = CI_analitico,
    df = df,
    vcov = V
  ))
}

RenRegMulti <- function(data, xname = "N", covariates = NULL,
                        alpha = 0.05, boot = FALSE, B = 1000, seed = 123) {
  # data: data.frame con columnas DESEMBARQUE y xname y covariates si los hay
  # xname: nombre (string) de la variable esfuerzo, p.ej. "N"
  # covariates: vector de nombres (strings) de variables auxiliares, p.ej. c("MES")
  #   las variables en 'covariates' se incluirán como factores automáticamente
  # boot: si TRUE, calcula IC bootstrap (percentil) con B réplicas
  # devuelve lista invisible con resultados y muestra un resumen tipo RenTram
  
  # chequeos
  if (!("DESEMBARQUE" %in% names(data))) stop("data debe tener columna 'DESEMBARQUE'")
  if (!(xname %in% names(data))) stop(paste0("data debe tener columna '", xname, "'"))
  n <- nrow(data)
  
  # construir fórmula
  formula_parts <- xname
  if (!is.null(covariates)) {
    # convertir covariates a factor en la fórmula
    cov_terms <- sapply(covariates, function(v) paste0("factor(", v, ")"))
    formula_parts <- c(formula_parts, cov_terms)
  }
  fmla <- as.formula(paste("DESEMBARQUE ~", paste(formula_parts, collapse = " + ")))
  
  # ajustar modelo
  m <- lm(fmla, data = data)
  coef_hat <- coef(m)
  p <- length(coef_hat)
  
  # matriz de diseño y sumas de columnas (coincide con coef names)
  MM <- model.matrix(m)       # n x p
  sums_cols <- colSums(MM)    # vector longitud p, nombre coincide con coef names
  sum_x <- sum(data[[xname]])
  
  # vector 'a' para delta method: a_j = sum(column_j) / sum_x
  a_vec <- as.numeric(sums_cols / sum_x)
  names(a_vec) <- names(coef_hat)
  
  # cov matrix of coef
  Vcoef <- vcov(m)            # p x p
  # estimated U and var(U)
  U_hat <- sum(sums_cols * coef_hat) / sum_x     # = t(a_vec) %*% coef_hat
  varU_hat <- as.numeric(t(a_vec) %*% Vcoef %*% a_vec)
  seU_hat <- sqrt(varU_hat)
  
  # degrees freedom and t critical
  df <- n - p
  tcrit <- qt(1 - alpha/2, df = df)
  CI_analitico <- U_hat + c(-1,1) * tcrit * seU_hat
  
  # bootstrap opcional: remuestrear filas, recalcular (coef, sums, U_boot)
  CI_boot <- NULL
  meanUboot <- sdUboot <- NA
  if (boot) {
    set.seed(seed)
    Uboot <- numeric(B)
    for (b in seq_len(B)) {
      i <- sample.int(n, size = n, replace = TRUE)
      dboot <- data[i, , drop = FALSE]
      mboot <- lm(fmla, data = dboot)
      # recomputar sums_cols_boot (usar model.matrix de la muestra bootstrap)
      MMb <- model.matrix(mboot)
      # Es importante alinear nombres: si alguna categoría no aparece en bootstrap,
      # las columnas pueden cambiar. Para robustez, usaremos coefs y sums según mm colnames.
      coefs_b <- coef(mboot)
      sums_b <- colSums(MMb)
      sum_x_b <- sum(dboot[[xname]])
      Uboot[b] <- sum(sums_b * coefs_b) / sum_x_b
    }
    CI_boot <- quantile(Uboot, probs = c(alpha/2, 1 - alpha/2), na.rm = TRUE)
    meanUboot <- mean(Uboot, na.rm = TRUE)
    sdUboot <- sd(Uboot, na.rm = TRUE)
  }
  
  # impresión tipo RenTram
  cat("========================================\n")
  cat("  ESTIMADOR DE REGRESIÓN MULTIPLE (Cochran extendido)\n")
  cat("========================================\n\n")
  cat("Fórmula ajustada:\n")
  cat(deparse(fmla), "\n\n")
  
  cat("Coeficientes estimados (coef_hat):\n")
  print(round(coef_hat, 6))
  cat("\nMatriz de var-cov de coeficientes (vcov):\n")
  print(round(Vcoef, 4))
  cat("\nSumas de columnas de la matriz de diseño (s_j):\n")
  print(round(sums_cols, 4))
  cat("\nVector a (s_j / sum(x)):\n")
  print(round(a_vec, 6))
  cat("\nEstimación del rendimiento:\n")
  cat(sprintf("  U estimado (kg jaiba/%s): %.6f\n", xname, U_hat))
  cat(sprintf("  Var(U estimado): %.6f\n", varU_hat))
  cat(sprintf("  se(U): %.6f\n\n", seU_hat))
  cat(sprintf("  IC analítico %.1f%% = [%.6f, %.6f] (t, df=%d)\n",
              100*(1-alpha), CI_analitico[1], CI_analitico[2], df))
  if (boot) {
    cat(sprintf("  IC bootstrap (percentil %.1f%%) = [%.6f, %.6f]\n",
                100*(1-alpha), CI_boot[1], CI_boot[2]))
    cat(sprintf("  mean(Uboot)=%.6f  sd(Uboot)=%.6f (B=%d)\n", meanUboot, sdUboot, B))
  }
  cat("\n========================================\n")
  
  # devolver resultados
  invisible(list(
    model = m,
    coef = coef_hat,
    vcov = Vcoef,
    sums_cols = sums_cols,
    a = a_vec,
    U = U_hat,
    varU = varU_hat,
    seU = seU_hat,
    CI_analitico = CI_analitico,
    CI_boot = CI_boot,
    Uboot_mean = meanUboot,
    Uboot_sd = sdUboot,
    df = df
  ))
}
